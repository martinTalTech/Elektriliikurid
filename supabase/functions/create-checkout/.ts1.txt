import { serve } from 'https://deno.land/std@0.177.0/http/server.ts'
import Stripe from 'https://esm.sh/stripe@14'

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') || '', { apiVersion: '2024-06-20' })

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', {
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Headers': 'authorization, content-type',
      }
    })
  }

  const { listing_id, plan } = await req.json()
  const amount = plan === 'premium' ? 599 : 299

  const session = await stripe.checkout.sessions.create({
    payment_method_types: ['card'],
    line_items: [{
      price_data: {
        currency: 'eur',
        product_data: {
          name: plan === 'premium' ? 'Premium kuulutus (60 päeva)' : 'Standard kuulutus (30 päeva)'
        },
        unit_amount: amount,
      },
      quantity: 1,
    }],
    mode: 'payment',
    success_url: `https://martintaltech.github.io/Elektriliikurid/#/kuulutused/${listing_id}?success=1`,
    cancel_url: `https://martintaltech.github.io/Elektriliikurid/#/lisa-kuulutus`,
    payment_intent_data: {
      metadata: { listing_id, plan }
    }
  })

  return new Response(JSON.stringify({ url: session.url }), {
    headers: {
      'Content-Type': 'application/json',
      'Access-Control-Allow-Origin': '*'
    }
  })
})
